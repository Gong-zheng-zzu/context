## 测试脚本模板

### 完整端到端测试脚本
```python
#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""
完整的端到端测试脚本
测试：登录 → 发送敏感信息 → 验证AI回复脱敏 → 验证存储
"""
import requests
import json
import time

BASE_URL = "http://localhost:8088/api"

def test_e2e_sensitive_redaction():
    print("=" * 80)
    print("端到端测试：敏感信息脱敏")
    print("=" * 80)
    
    # 步骤1: 登录
    print("\n[步骤1] 登录获取JWT Token")
    login_resp = requests.post(
        f"{BASE_URL}/role/login",
        json={
            "role": "caregiver",
            "user_id": "test_caregiver_001",
            "password": "test123",
            "workspace_id": "default"
        }
    )
    token = login_resp.json()['data']['token']
    print(f"✓ 登录成功，Token: {token[:20]}...")
    
    headers = {
        "Authorization": f"Bearer {token}",
        "Content-Type": "application/json"
    }
    
    # 步骤2: 发送包含敏感信息的消息
    print("\n[步骤2] 发送包含敏感信息的消息")
    test_message = "张奶奶的电话是13812345678，身份证是110101199001011234，银行卡是6222021234567890123"
    
    chat_resp = requests.post(
        f"{BASE_URL}/chat",
        json={
            "user_id": "test_caregiver_001",
            "session_id": f"test_session_{int(time.time())}",
            "message": test_message,
            "history": []
        },
        headers=headers
    )
    
    ai_response = chat_resp.json()['response']
    print(f"✓ AI回复: {ai_response}")
    
    # 步骤3: 验证AI回复中的敏感信息已脱敏
    print("\n[步骤3] 验证敏感信息脱敏")
    
    checks = [
        ("电话号码", "13812345678", "138****5678"),
        ("身份证号", "110101199001011234", "110***********1234"),
        ("银行卡号", "6222021234567890123", "622202*********0123")
    ]
    
    all_passed = True
    for name, original, redacted in checks:
        if original in ai_response:
            print(f"✗ {name} 未脱敏: 发现原始值 {original}")
            all_passed = False
        elif redacted in ai_response:
            print(f"✓ {name} 已脱敏: {redacted}")
        else:
            print(f"? {name} 状态未知")
    
    # 步骤4: 验证安全扫描API
    print("\n[步骤4] 验证安全扫描API")
    scan_resp = requests.post(
        f"{BASE_URL}/security/scan",
        json={"content": test_message},
        headers=headers
    )
    
    scan_result = scan_resp.json()
    print(f"✓ 检测到 {scan_result['count']} 个敏感信息")
    for info in scan_result['sensitive_infos']:
        print(f"  - {info['type']}: {info['value']} (置信度: {info['confidence']})")
    
    print("\n" + "=" * 80)
    if all_passed:
        print("✓ 测试通过：所有敏感信息已正确脱敏")
    else:
        print("✗ 测试失败：部分敏感信息未脱敏")
    print("=" * 80)

if __name__ == "__main__":
    test_e2e_sensitive_redaction()
```
